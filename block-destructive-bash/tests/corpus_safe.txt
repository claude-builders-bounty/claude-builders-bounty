# Safe commands -- every line MUST pass through untouched
# (exit 0, no JSON on stdout, no log entry).
# One command per line; blank lines and lines starting with # are ignored.

# --- rm: the classic near-misses ---
rm -f file.txt
rm file.txt
rm -v file.txt
rm -r build
rm -- -rf

# --- git: ordinary pushes ---
git push origin main
git push -u origin feature/login
git push --tags
git push --dry-run origin main
git -C /repo push origin main

# --- SQL: targeted deletes and reads via a client ---
mysql -e "DELETE FROM t WHERE id=1"
psql -c "DELETE FROM sessions WHERE expires_at < now()"
mysql -e "SELECT * FROM users WHERE id=1"

# --- dangerous words that are only *mentioned*, never executed ---
echo "rm -rf is dangerous"
echo 'Never run DROP TABLE in prod'
grep -rn "DROP TABLE" migrations/
grep -rn "DELETE FROM" src/
cat docs/rm-guide.md
printf 'rm -rf danger\n'

# --- the SQL-looking shell utility, not a database ---
truncate -s 0 app.log

# --- everyday commands ---
ls -la && cat file.txt
npm test
python3 script.py
git status
git commit -m "remove -rf flag handling"
