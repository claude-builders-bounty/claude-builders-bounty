```python
#!/usr/bin/env python3
"""
Opire: Next.js + SQLite SaaS Specification
A Pythonic container for an opinionated CLAUDE.md configuration.

Usage:
  # To generate the raw content for CLAUDE.md:
  $ python -c "from spec import SPEC; print(SPEC.render())"

  # Or run this file directly:
  $ python spec.py
"""

from dataclasses import dataclass, field
from typing import List, Optional

@dataclass
class NextJSSpec:
    title: str = "SaaS Project"
    description: str = "An opinionated guide for a Next.js 15 App Router + SQLite SaaS."
    stack_versions: dict = field(default_factory=lambda: {
        "next": "15.0.x",
        "typescript": "5.5.x",
        "sqlite": "better-sqlite3 / Turso",
        "ts-node": "9.5.x"
    })
    folder_structure: List[str] = field(default_factory=lambda: [
        "/app (Next.js App Router)",
        "/db (SQLite logic & schema)",
        "/lib (Shared utility types & hooks)",
        "/components (Pure Server Components)",
        "/public (Static assets)",
        "/types (Type definitions from DB)"
    ])
    dev_commands: List[str] = field(default_factory=lambda: [
        "dev --turbopack (Fastest HMR)",
        "db:seed (Seed initial data)",
        "db:sync (Sync schema from files)"
    ])
    
    # The "Opinionated" Section Content
    content: dict = field(default_factory=lambda: {
        "sections": [
            {
                "name": "Stack & Versions",
                "rules": [
                    "Use `next` version 15 for App Router default exports.",
                    "Prefer `better-sqlite3` over raw `sqlite` for synchronous file-based DB performance.",
                    "Ensure `lib/*` uses `async/await` to bridge JS & DB sync nature."
                ]
            },
            {
                "name": "Folder Structure & Naming",
                "rules": [
                    "**Pages:** Use `/app/` with file-based routing (`[id]/page.ts`).",
                    "**Components:** Group by feature (`/components/dashboard/` vs `/components/ui/`).",
                    "**Files:** Prefix feature folders with noun (`/features/user/`) vs action (`/actions/`)."
                ]
            },
            {
                "name": "SQL / Migration Conventions",
                "rules": [
                    "**Schema:** Define tables as `db` class instances in `/db/table_name.ts`.",
                    "**Migrations:** Use filename suffix `-migrations.ts` in `/db/` folder.",
                    "**Transactions:** Wrap writes in `db.transaction()` for batched DB ops.",
                    "**Indexes:** Add indexes immediately (`db.exec("CREATE INDEX...")`) after creation."
                ]
            },
            {
                "name": "Component Patterns",
                "rules": [
                    "**Server Default:** Components default to Server Components (`'use client'` only when needed).",
                    **"Props First:** Always spread props (`{...props}`) to accept arbitrary UI tokens.",
                    **"Loading States:** Wrap logic in `useEffect` and render `skeleton` loaders.",
                    **"Data Fetching:** Use `async` functions at top level for Server-side fetching."
                ]
            },
            {
                "name": "What We Don't Do (Anti-Patterns)",
                "rules": [
                    "**Avoid:** Inline SQL strings. Use typed `Table` objects.",
                    "**Avoid:** Too many `useEffect` in Server Components (preface `use client`).",
                    "**Avoid:** Stale props. Pass callbacks instead of direct DB objects.",
                    "**Avoid:** Unprefixed `useId` on Server. Use `@useId` from `lib/ui`."
                ]
            }
        ]
    })

    def render(self) -> str:
        """Renders the Markdown string ready for CLAUDE.md."""
        
        def render_list(lst: List[str]) -> str:
            return "\n".join([f"- {item}" for item in lst])
        
        def render_section(section: dict) -> str:
            return f"### {section['name']}\n{render_list(section['rules'])}\n"
            
        output = f"""# 💰 **CLAUDE.md**: SaaS Context
## 💰 Bounty: $75 — powered by [Opire](https://opire.dev)

## Description
An opinionated, production-ready guide for a typical SaaS project built with **Next.js 15 App Router** and **SQLite** (better-sqlite3 or Turso).

---
## Acceptance Criteria
- [ ] Covers: project structure, naming conventions, DB migration rules
- [ ] Includes: dev commands, patterns to follow, anti-patterns to avoid
- [ ] Opinionated — not generic. Every rule has a reason.
- [ ] Usable without modification on a greenfield Next.js + SQLite project

---"""

        # Stack Header
        output += f"""## Stack & Versions
| Service | Version | Note |
| :-- | :-- | :-- |
| **Next.js** | `{self.stack_versions['next']}` | App Router defaults |
| **TypeScript** | `{self.stack_versions['typescript']}` | Strict mode |
| **SQLite** | `{self.stack_versions['sqlite']}` | File-based sync/async |

### Folder Structure
```bash
/app (Server Components)
  ├── [slug] (Dynamic Routes)
  ├── globals.css
/db (Better-sqlite3 schema)
/lib (Utility & DB wrappers)
  ├── types (Zod/DB types)
/components (Pure UI)
```

---
## SQL & Migration Conventions
**The DB Rule:** Treat the `db` instance as the source of truth.

1. **Schema Definition:** 
   - Use `/db/users.ts` instead of flat schema files.
   - Wrap reads in `session.get()` for transaction safety.
   
2. **Migration Naming:**
   - Files in `/db/migrations` named `timestamp_description`.
   - Run `db.sync()` to apply schema changes instantly.

**Naming Rules:**
- **Folders:** Noun-first (`/features/`, `/pages/`, `/actions/`).
- **Files:** Use lowercase or PascalCase depending on context (Feature folder = PascalCase).
- **Variables:** `snake_case` for DB names, `camelCase` for App logic.

---
## Component Patterns
**Goal:** Zero boilerplate Server Components.

1. **The Wrapper Pattern:**
   ```tsx
   // @/components/ui/Card.tsx
   'use client' // Only where state is needed
   
   export function Card({ children, className }) {
     return (
       <div className={`p-6 bg-white rounded shadow ${className}`}>
         {children}
       </div>
     )
   }
   ```

2. **Async Fetching:**
   - Default to `async` at the top level.
   - Use `use()` for loading states (e.g., `const user = use(userQuery)`).

---
## What We Don't Do (Anti-Patterns)
1. **Inline SQL:** Avoid raw strings if types exist.
2. **Stale Props:** Wrap `db.rows` in `useMemo` or callback functions.
3. **Unprefixed IDs:** Always use `id` on DB objects, `useId()` in UI.
4. **Over-normalized DB:** SaaS needs speed. Keep tables flat, add a view for deep joins.

---
## Dev Commands
```bash
# Fastest HMR
next dev --turbo

# Seed DB
npm run db:seed

# Typecheck DB specifically
npm run types:check
```

> **💡 Pro Tip:** Ensure your DB is initialized in `app/layout.tsx` via a provider wrapper for global DB state access."""

        return output

# Allow direct printing if run as script
if __name__ == "__main__":
    spec = NextJSSpec()
    # Clean up the output for copy-paste or direct file import
    print(spec.render())
