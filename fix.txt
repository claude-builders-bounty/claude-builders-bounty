#!/usr/bin/env python3
"""
Opire / Claude Code Pre-Tool-Use Hook
Intercepts dangerous bash commands before execution and logs them.

Install:
1. mkdir -p ~/.claude/hooks
2. cp this_file ~/.claude/hooks/hook.py
"""

import sys
import os
import re
from datetime import datetime
from pathlib import Path

# --- Configuration ---
HOOK_SCRIPT_PATH = Path(__file__).resolve()
HOOKS_DIR = HOOK_SCRIPT_PATH.parent
LOG_FILE = HOOKS_DIR / "blocked.log"

# Patterns to match destructive commands (Case Insensitive for safety)
DANGEROUS_PATTERNS = [
    r"(rm\s+-rf)",      # e.g., rm -rf /
    r"(DROP\s+TABLE\s+)", # e.g., DROP TABLE users
    r"(git\s+push\s+--force)", # e.g., git push --force origin main
    r"(TRUNCATE\s+\w+)",    # e.g., TRUNCATE table
    r"(DELETE\s+FROM\s+\w+\s+WHERE)", # e.g., DELETE FROM users WHERE (id)
]

def get_project_path():
    # Determine where the project is currently running from
    cwd = os.getcwd()
    # Clean up cwd to match standard naming conventions
    return cwd

def log_blocking_attempt(command):
    """Appends the blocked command to the persistent log file."""
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    project_path = get_project_path()
    
    # Ensure log directory exists
    LOG_FILE.parent.mkdir(parents=True, exist_ok=True)
    
    log_entry = f"{timestamp} | {project_path} | {command}\n"
    with open(LOG_FILE, "a") as f:
        f.write(log_entry)

def is_dangerous(cmd):
    """Checks if the command matches any of the dangerous patterns."""
    # Clean command for matching (handle quotes/whitespace)
    clean_cmd = re.sub(r'[\'"]', '', cmd)
    
    for pattern in DANGEROUS_PATTERNS:
        if re.search(pattern, clean_cmd, re.IGNORECASE):
            return True
    return False

def main():
    if len(sys.argv) < 2:
        # If no arguments passed, it might be running as a standalone script
        # Echo the command or simply pass through
        pass

    # Join all arguments after the script name itself
    command = " ".join(sys.argv[1:])
    
    # If the first arg was the path to this hook, strip it for cleaner logic 
    # (Though sys.argv usually handles it well)
    
    # Check if command is "dangerous"
    if command and is_dangerous(command):
        # Log the attempt
        log_blocking_attempt(command)
        # Print a clear message for the LLM to understand context
        print(f"🔍 Interception: {command} (See ~/.claude/hooks/blocked.log)")
        
    else:
        # Standard output for normal commands
        if command:
            print(f"📤 Execution: {command}")

if __name__ == "__main__":
    # Ensure the script runs the logic
    if len(sys.argv) > 0:
        main()
    else:
        # Fallback if called without explicit args (handles stdin)
        main()
