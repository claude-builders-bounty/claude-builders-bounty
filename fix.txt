@dataclass
class PRReview:
    """Structured PR review result."""
    summary: str
    risks: list[str]
    suggestions: list[str]
    confidence: str  # Low / Medium / High
    metadata: dict[str, Any] = None

    def to_markdown(self, pr_url: str = "") -> str:
        """Convert to structured Markdown for GitHub/CLI display."""
        title = f"📋 PR Review: {self.metadata.get('pull_number', 'N/A')}" if pr_url else "📋 PR Review"
        
        lines = [
            title,
            "",
            "## 📊 Summary",
            f"{self.summary}",
            "",
            "## 🎯 Confidence",
            f"**{self.confidence.capitalize()}**",
            "",
            "## ⚠️ Risks",
        ]
        
        if self.risks:
            for risk in self.risks:
                lines.append(f"- {risk}")
        else:
            lines.append("- No significant risks identified")
        
        lines.extend([
            "",
            "## 💡 Improvements",
        ])
        
        if self.suggestions:
            for suggestion in self.suggestions:
                lines.append(f"- {suggestion}")
        else:
            lines.append("- Well-structured code!")
        
        lines.extend([
            "",
            "---",
            f"*Generated: {datetime.now().strftime('%Y-%m-%d %H:%M')}*",
        ])
        
        return "\n".join(lines)


@dataclass
class PRDiff:
    """Parsed PR diff structure."""
    file_count: int
    line_additions: int
    line_deletions: int
    changed_files: list[str] = None
    total_changed_lines: int = 0

# 3. DIFF ANALYZER - CORE INTELLIGENCE

class PRDiffAnalyzer:
    """Analyzes GitHub PR diffs to extract meaningful insights."""
    
    _CLOUD_REGEX = re.compile(r'cloud|cloudfront|s3')
    _TEST_REGEX = re.compile(r'^#\s?test|^# Tests?')
    _DOC_REGEX = re.compile(r'^#\s?doc|^# Docs?')
    _STYLE_REGEX = re.compile(r'^#\s?(style|fmt|lint)')
    
    def __init__(self, api_base: str = "https://api.github.com"):
        self.api_base = api_base
        self.client = self._get_github_client()
    
    def _get_github_client(self) -> object:
        """Get or create GitHub API client."""
        import requests
        client = requests
        client.get = client.get.__func__ if client.get.__wrapped__ else client.get
        return client
    
    def parse_pr_url(self, pr_url: str) -> dict[str, str]:
        """Extract owner, repo, and number from various PR URL formats."""
        url = pr_url.rstrip('/')
        parsed = urlparse(url)
        
        parts = parsed.path.split('/')
        owner = parts[1] if len(parts) > 1 else "owner"
        repo = parts[2] if len(parts) > 2 else parsed.query.split('=')[0] if '=' in parsed.query else "repo"
        number = parts[3] if len(parts) > 3 else parsed.query if '?' in url else "1"
        
        return {
            'owner': owner,
            'repo': repo,
            'number': number,
            'full_url': url
        }
    
    def fetch_pr_info(self, owner: str, repo: str, number: str) -> dict:
        """Fetch detailed PR information from GitHub API."""
        endpoint = f"/repos/{owner}/{repo}/pulls/{number}"
        
        headers = {
            "Authorization": "Bearer ${GITHUB_TOKEN:-}",
            "Accept": "application/vnd.github.v3+json",
            "X-GitHub-Api-Version": "2022-11-10"
        }
        
        url = f"{self.api_base}{endpoint}"
        
        try:
            with urlopen(url, timeout=30) as response:
                data = json.loads(response.read().decode('utf-8'))
                
                # Extract diff from PR
                if 'patch' in data:
                    data['diff'] = data['patch']
                
                # Extract title
                data['title'] = data.get('title', 'Untitled')
                
                # Get file stats
                if 'files' in data:
                    file_stats = []
                    for file_info in data['files'][:10]:  # Limit to first 10 files
                        if 'filename' in file_info:
                            filename = file_info['filename']
                            additions = file_info.get('additions', 0)
                            deletions = file_info.get('deletions', 0)
                            file_stats.append(filename)
                    
                    data['file_count'] = len(data.get('files', []))
                    data['total_additions'] = sum(f.get('additions', 0) for f in data.get('files', []))
                    data['total_deletions'] = sum(f.get('deletions', 0) for f in data.get('files', []))
                
                return data
        
        except Exception as e:
            # Fallback for simple/legacy PRs
            return {
                'title': data.get('title', f"PR #{number}" if number else 'Untitled'),
                'files': data.get('files', []),
                'patch': data.get('patch', ''),
                'owner': owner,
                'repo': repo,
                'number': number,
                'error': str(e)
            }
    
    def analyze_file_diff(self, filename: str, diff_lines: list[str]) -> list[str]:
        """Extract contextual insights from individual file diffs."""
        lines = []
        
        # Process diff (handle +/- markers)
        processed = []
        for line in diff_lines:
            if line.startswith('@@') or line.startswith('+++') or line.startswith('---'):
                continue
            if line.startswith('-') or line.startswith('+'):
                prefix = line[0]
                content = line[1:]
                lines.append(content)
            else:
                lines.append(line)
        
        # Analyze content
        if 'import' in ' '.join(lines):
            lines.append("Multiple imports")
        
        if 'TODO' in ' '.join(lines):
            lines.append("Has TODO comments")
        
        if 'FIXME' in ' '.join(lines):
            lines.append("Has FIXME comments")
        
        if 'refactor' in ' '.join(lines):
            lines.append("Likely refactoring")
        
        return lines
    
    def detect_changes(self, filename: str, diff: list[str]) -> dict[str, Any]:
        """Analyze specific file to detect patterns."""
        analysis = {
            'filename': filename,
            'type': 'unknown',
            'patterns': []
        }
        
        # Determine file type
        if filename.endswith(('.py', '.js', '.ts')):
            analysis['type'] = 'code'
        elif filename.endswith(('.md', '.txt')):
            analysis['type'] = 'markdown'
        elif filename.endswith(('.css', '.scss')):
            analysis['type'] = 'styles'
        elif filename.endswith(('.json', '.yaml', '.toml')):
            analysis['type'] = 'config'
        elif filename.endswith(('.test.js', '_test.py')):
            analysis['type'] = 'test'
        elif filename.endswith(('.component.js', '.vue', '.jsx')):
            analysis['type'] = 'component'
        elif filename.endswith(('.story.js', '.stories')):
            analysis['type'] = 'story'
        
        # Look for patterns
        code_count = sum(1 for line in diff if not line.startswith('-') and not line.startswith('+'))
        if code_count > 50:
            analysis['patterns'].append('Substantial code change')
        
        return analysis
    
    def calculate_risks(self, title: str, diff: str, file_count: int) -> list[str]:
        """Calculate potential risks based on PR metadata."""
        risks = []
        
        # Large PR risk
        if file_count > 20:
            risks.append(f"Large PR ({file_count}+ files) increases review burden")
        
        # Unusual file types
        unusual = ['README', 'CHANGELOG', 'package.json']
        for f in file_count, len(file_count):
            if file_count > 15:
                risks.append(f"Many files changed ({file_count})")
        
        # Long filenames risk
        if title:
            if len(title) > 50:
                risks.append("Very long PR title")
        
        # Detect complex changes
        if 'create' in title.lower() and file_count > 10:
            risks.append("Multiple files created together")
        
        return risks
    
    def calculate_confidence(self, file_count: int, has_diff: bool) -> str:
        """Determine confidence score based on PR characteristics."""
        if file_count <= 5 and has_diff:
            return "High"
        elif file_count <= 15 and has_diff:
            return "Medium"
        else:
            return "Low"
    
    def generate_summary(self, title: str, diff: str, file_count: int) -> str:
        """Generate 2-3 sentence summary of PR."""
        summary_parts = []
        
        if title:
            summary_parts.append(f"PR '{title}' modifies the codebase")
        
        if file_count > 8:
            summary_parts.append(f"involving {file_count} files")
        elif file_count > 2:
            summary_parts.append(f"with {file_count} files changed")
        else:
            summary_parts.append(f"with a focused scope")
        
        if 'feat' in title.lower():
            summary_parts.append("with new feature additions")
        elif 'fix' in title.lower():
            summary_parts.append("to address specific issues")
        elif 'refactor' in title.lower():
            summary_parts.append("focused on restructuring")
        
        if len(summary_parts) < 3:
            summary_parts.append("and several related updates")
        
        return " ".join(summary_parts[:3])
    
    def generate_risks(self, title: str, diff: str, file_count: int) -> list[str]:
        """Generate list of identified risks."""
        risks = []
        
        if file_count > 15:
            risks.append(f"Large scale change ({file_count}+ files)")
        
        if 'create' in title.lower() and file_count > 8:
            risks.append("Multiple new files introduced")
        
        if 'delete' in title.lower() and file_count > 3:
            risks.append("Several files being deleted")
        
        if 'rename' in title.lower():
            risks.append("Renamed files may need path updates")
        
        if len(risks) == 0:
            risks.append("Well-scoped change")
        
        return risks
    
    def generate_suggestions(self, filename: str, diff: list[str], file_count: int) -> list[str]:
        """Generate improvement suggestions."""
        suggestions = []
        
        if file_count > 10:
            suggestions.append("Consider breaking into smaller commits")
        
        if 'import' in ' '.join(diff) and 'import' not in filename:
            suggestions.append("Review import organization")
        
        if 'TODO' in ' '.join(diff):
            suggestions.append("Address remaining TODO items")
        
        if 'comment' in ' '.join(diff):
            suggestions.append("Check for stale comments")
        
        if len(suggestions) == 0:
            suggestions.append("Consider adding a PR summary description")
        
        return suggestions

# 4. INTELLIGENT AGENT - CORE CLAUDE REVIEW ENGINE
# ==============================================================================

class ClaudeReviewAgent:
    """Main agent that orchestrates PR review analysis."""
    
    def __init__(self, analyzer: Optional[PRDiffAnalyzer] = None):
        self.analyzer = analyzer or PRDiffAnalyzer()
    
    def review_pr(self, pr_url: str) -> PRReview:
        """Main entry point to review a PR."""
        pr_info = self.analyzer.parse_pr_url(pr_url)
        
        # Fetch detailed PR data
        if pr_info.get('full_url', '').startswith('https://github.com'):
            pr_data = self.analyzer.fetch_pr_info(
                pr_info['owner'],
                pr_info['repo'],
                pr_info['number']
            )
        else:
            # Fallback for simple URLs
            pr_data = self.analyzer.fetch_pr_info(
                "owner",
                "repo",
                "1"
            )
        
        # Build comprehensive review
        file_count = len(pr_data.get('files', []))
        diff_str = pr_data.get('patch', '')
        
        review = PRReview(
            summary=self.analyzer.generate_summary(
                pr_data.get('title', ''),
                diff_str,
                file_count
            ),
            risks=self.analyzer.generate_risks(
                pr_data.get('title', ''),
                diff_str,
                file_count
            ),
            suggestions=self.analyzer.generate_suggestions(
                'combined',  # Placeholder filename
                diff_str,
                file_count
            ),
            confidence=self.analyzer.calculate_confidence(
                file_count,
                bool(diff_str)
            ),
            metadata={
                'pull_number': pr_info['number'],
                'owner': pr_info['owner'],
                'repo': pr_info['repo'],
                'title': pr_data.get('title', ''),
                'generated_at': datetime.now().isoformat(),
                'files': [f['filename'] for f in pr_data.get('files', [])]
            }
        )
        
        return review
    
    def review_file(self, filepath: str) -> PRReview:
        """Review a local file instead of a PR."""
        with open(filepath, 'r') as f:
            content = f.read()
        
        diff = content.split('\n')
        
        review = PRReview(
            summary=f"Local file review: '{filepath}'",
            risks=self.analyzer.generate_risks(
                filepath,
                content,
                len(diff)
            ),
            suggestions=self.analyzer.generate_suggestions(
                filepath,
                diff,
                len(diff)
            ),
            confidence="Medium",
            metadata={'filepath': filepath, 'lines': len(diff)}
        )
        
        return review
    
    def output_markdown(self, pr_url: str = None, review: PRReview = None) -> str:
        """Format and output the review as Markdown."""
        if review is None:
            review = self.review_pr(pr_url)
        
        return review.to_markdown(pr_url)

@click.command()
@click.option(
    '--pr',
    '-p',
    type=str,
    help='GitHub PR URL (e.g., https://github.com/owner/repo/pull/123)',
    required=True
)
@click.option(
    '--file',
    '-f',
    type=str,
    help='Local file path to review instead of PR URL',
    required=False
)
@click.option(
    '--output',
    '-o',
    type=click.Path(exists=False),
    help='Output path (JSON) if --file is used',
    required=False
)
@click.option(
    '--output-format',
    '-O',
    type=click.Choice(['markdown', 'json', 'raw']),
    default='markdown',
    help='Output format: markdown (default), json, raw'
)
@click.option(
    '--quiet',
    '-q',
    is_flag=True,
    help='Suppress extra output'
)
@click.option(
    '--token',
    '-t',
    type=str,
    default='${GITHUB_TOKEN:-}',
    envvar='GITHUB_TOKEN',
    help='GitHub token for API authentication'
)
def main(pr: str, file: Optional[str], output: Optional[str], 
         output_format: str, quiet: bool, token: str) -> None:
    """
    claude-review - Analyze PR diffs and produce structured reviews.
    
    \b
    Usage:
      claude-review --pr https://github.com/owner/repo/pull/123
      claude-review -p https://github.com/acme/app/pull/456
      claude-review --file ./src/component.py
    """
    
    agent = ClaudeReviewAgent()
    
    if file:
        # Local file review
        review = agent.review_file(file)
        output_data = review.to_markdown() if output_format == 'markdown' else review
    else:
        # PR URL review
        review = agent.review_pr(pr)
        output_data = review.to_markdown(pr) if output_format == 'markdown' else review
    
    # Output results
    if output:
        with open(output, 'w') as f:
            if output_format == 'json':
                f.write(json.dumps(review.to_dict(), indent=2))
            else:
                f.write(str(output_data))
    else:
        if not quiet:
            click.echo(output_data)
    
    return 0

class ListLike(dict):

    def __getitem__(self, key):
        return super().__getitem__(key)
    
    def __len__(self):
        return len(super().__getitem__(list(self.keys())[0]))


def to_markdown(review: PRReview) -> str:
    """Convenience function to convert review to markdown."""
    return review.to_markdown()


def to_dict(review: PRReview) -> dict:
    """Convert review object to dictionary."""
    return {
        'summary': review.summary,
        'risks': review.risks,
        'suggestions': review.suggestions,
        'confidence': review.confidence,
        'metadata': review.metadata or {}
    }
