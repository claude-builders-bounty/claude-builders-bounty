# Solution for: [BOUNTY $50] SKILL: Generate a structured CHANGELOG from git history


**`changelog.sh`**
```sh
#!/usr/bin/env bash
# changelog.sh — Generate a structured CHANGELOG.md from git history
# Usage: bash changelog.sh [version]
# Example: bash changelog.sh 1.2.0

set -euo pipefail

VERSION="${1:-Unreleased}"
OUTPUT_FILE="CHANGELOG.md"
DATE=$(date +%Y-%m-%d)

# 1. Find the last git tag
LAST_TAG=$(git describe --tags --abbrev=0 2>/dev/null || echo "")

if [ -z "$LAST_TAG" ]; then
  echo "No git tags found - using full commit history."
  COMMIT_RANGE=""
else
  echo "Generating changelog since tag: $LAST_TAG"
  COMMIT_RANGE="${LAST_TAG}..HEAD"
fi

# 2. Fetch commit subjects in that range (skip merge commits)
if [ -z "$COMMIT_RANGE" ]; then
  COMMITS=$(git log --no-merges --pretty=format:"%s" 2>/dev/null || echo "")
else
  COMMITS=$(git log "$COMMIT_RANGE" --no-merges --pretty=format:"%s" 2>/dev/null || echo "")
fi

if [ -z "$COMMITS" ]; then
  echo "No commits found since last tag. Nothing to changelog."
  exit 0
fi

# 3. Categorize commits
ADDED=()
FIXED=()
CHANGED=()
REMOVED=()

while IFS= read -r commit; do
  [ -z "$commit" ] && continue

  # Normalize to lowercase for matching
  lower=$(echo "$commit" | tr '[:upper:]' '[:lower:]')

  # Strip conventional commit prefix for display (e.g. "feat: " becomes "")
  display=$(echo "$commit" | sed -E 's/^[a-zA-Z]+(\([^)]+\))?!?: //')

  if echo "$lower" | grep -qE '^(feat|add|new|creat)[:\( ]|^(add|new|creat)\b'; then
    ADDED+=("$display")
  elif echo "$lower" | grep -qE '^(fix|bug|patch|resolv|hotfix)[:\( ]|^(fix|bug|patch|resolv|hotfix)\b'; then
    FIXED+=("$display")
  elif echo "$lower" | grep -qE '^(remov|delet|drop|deprecat)[:\( ]|^(remov|delet|drop|deprecat)\b'; then
    REMOVED+=("$display")
  else
    CHANGED+=("$display")
  fi
done <<< "$COMMITS"

# 4. Write CHANGELOG.md
{
  echo "# Changelog"
  echo ""
  echo "All notable changes to this project will be documented in this file."
  echo ""
  echo "The format is based on [Keep a Changelog](https://keepachangelog.com/en/1.0.0/),"
  echo "and this project adheres to [Semantic Versioning](https://semver.org/spec/v2.0.0.html)."
  echo ""
  echo "---"
  echo ""
  echo "## [$VERSION] - $DATE"
  echo ""

  if [ ${#ADDED[@]} -gt 0 ]; then
    echo "### Added"
    for item in "${ADDED[@]}"; do
      echo "- $item"
    done
    echo ""
  fi

  if [ ${#FIXED[@]} -gt 0 ]; then
    echo "### Fixed"
    for item in "${FIXED[@]}"; do
      echo "- $item"
    done
    echo ""
  fi

  if [ ${#CHANGED[@]} -gt 0 ]; then
    echo "### Changed"
    for item in "${CHANGED[@]}"; do
      echo "- $item"
    done
    echo ""
  fi

  if [ ${#REMOVED[@]} -gt 0 ]; then
    echo "### Removed"
    for item in "${REMOVED[@]}"; do
      echo "- $item"
    done
    echo ""
  fi

  echo "---"
  echo ""
  echo "*Generated by [generate-changelog](https://github.com/claude-builders-bounty/claude-builders-bounty) on $DATE*"
} > "$OUTPUT_FILE"

echo "CHANGELOG.md generated successfully."
if [ -n "$LAST_TAG" ]; then
  echo "Range: $LAST_TAG..HEAD -> version $VERSION"
fi
echo "Added: ${#ADDED[@]}  Fixed: ${#FIXED[@]}  Changed: ${#CHANGED[@]}  Removed: ${#REMOVED[@]}"

```
